function normalize(s) {
  return String(s || "")
    .toLowerCase()
    .replace(/[Ａ-Ｚａ-ｚ０-９]/g, ch =>
      String.fromCharCode(ch.charCodeAt(0) - 0xfee0))
    .replace(/[ァ-ヶ]/g, ch =>
      String.fromCharCode(ch.charCodeAt(0) - 0x60))
    .replace(/\s+/g, "");
}

function kanaRow(kana) {
  const k = normalize(kana);
  if (!k) return "";
  const first = k.charAt(0);
  const rows = {

    "か":"か","き":"か","く":"か","け":"か","こ":"か",

function firstKanaGroup(item) {

  return kanaRow(item.kana) || kanaRow(item.name);
}

function uniqueSorted(key) {
  return [...new Set(DATA.map(x => x[key]).filter(Boolean))]
    .sort((a,b) => a.localeCompare(b, "ja"));
}

function fillSelect(el, values) {
  for (const value of values) {
    const option = document.createElement("option");
    option.value = value;
    option.textContent = value;
    el.appendChild(option);
  }
}

function makeKanaButtons() {
  KANA_GROUPS.forEach(([hiragana, katakana]) => {
    const btn = document.createElement("button");
    btn.type = "button";
    btn.textContent = hiragana;
    btn.dataset.kana = hiragana;
    btn.title = hiragana + "行を表示";
    btn.addEventListener("click", () => {
      selectedKana = selectedKana === hiragana ? "" : hiragana;
      document.querySelectorAll("#kanaGrid button").forEach(b =>
        b.classList.toggle("active", b.dataset.kana === selectedKana)
      );
      render();
    });
    kanaGrid.appendChild(btn);
  });
}

function esc(value) {
  return String(value || "").replace(/[&<>"']/g, c => ({
    "&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"
  })[c]);
}

function render() {
  const kw = normalize(keywordEl.value);
  const c1 = class1El.value;
  const c2 = class2El.value;

  const filtered = DATA.filter(item => {
    const allText = normalize([
      item.name, item.class1, item.class2, item.indication, item.dosage
    ].join(" "));
    const keywordOK = !kw || allText.includes(kw);
    const kanaOK = !selectedKana || firstKanaGroup(item) === selectedKana;
    const class1OK = !c1 || item.class1 === c1;
    const class2OK = !c2 || item.class2 === c2;
    return keywordOK && kanaOK && class1OK && class2OK;
  });

  countEl.textContent = `${filtered.length}件 / 全${DATA.length}件`;
  resultsEl.innerHTML = filtered.length
    ? filtered.map(card).join("")
    : '<div class="panel empty">該当する医薬品はありません。</div>';
}
